73inline Vec3 projectVector(const Vec3& world, const Mat4& view, const Mat4& projection, const float width, const float height)
74{
75 Vec4 res = world.homogeneous(); // add w=1 component
76 res = projection * view * res; // multiply by view and projection
77 Vec3 ndc = res.hnormalized(); // res / res.w() to Normalized Device Coords
78
79 Vec3 viewport;
80 viewport.x() = (ndc.x() + 1.0f) / 2.0f * width; // add 1.0f because ndc can be from -1 to 1
81 viewport.y() = (1.0f - ndc.y()) / 2.0f * height; // invert Y, as it points downwards on screens ¯\_(ツ)_/¯
82 viewport.z() = ndc.z(); // usefull data, for frustum culling for example...
83 return viewport;
84}

94inline Intersection intersect(const Ray& ray, const Sphere& sphere)
95{
96 Intersection res;
97 constexpr float tMin = 0.0f;
98 constexpr float tMax = std::numeric_limits<float>::max();
99
100 Eigen::Vector3f oc = ray.origin - sphere.center;
101
102 float a = ray.direction.squaredNorm();
103 float b = 2.0f * oc.dot(ray.direction);
104 float c = oc.squaredNorm() - sphere.radius * sphere.radius;
105
106 float discriminant = b * b - 4.0f * a * c;
107
108 if (discriminant < 0.0f)
109 {
110 res.hit = false;
111 return res;
112 }
113
114 float sqrtDisc = std::sqrt(discriminant);
115 float t = (-b - sqrtDisc) / (2.0f * a);
116 if (t < tMin || t > tMax)
117 {
118 t = (-b + sqrtDisc) / (2.0f * a);
119 if (t < tMin || t > tMax)
120 {
121 res.hit = false;
122 return res;
123 }
124 }
125
126 res.hit = true;
127 res.point = ray.origin + ray.direction * t;
128 res.distance = t;
129 return res;
130}

132inline Intersection intersect(const Ray& ray, const Plane& plane)
133{
134 Intersection res;
135 res.hit = false;
136 if (const float denominator = ray.direction.dot(plane.normal); denominator != 0)
137 {
138 const float distance = -(plane.normal.dot(ray.origin) + plane.distance) / denominator;
139
140 res.hit = distance > 0;
141 if (res.hit)
142 {
143 res.point = ray.origin + ray.direction * distance;
144 res.distance = distance;
145 }
146 }

152{
153 Intersection res;
154 bool insideBox = (ray.origin.x() > box.min.x()) && (ray.origin.x() < box.max.x()) && (ray.origin.y() > box.min.y()) &&
155 (ray.origin.y() < box.max.y()) && (ray.origin.z() > box.min.z()) && (ray.origin.z() < box.max.z());
156
157 if (insideBox) ray.direction = ray.direction * -1;
158
159 float t[11] = {0};
160
161 t[8] = 1.0f / ray.direction.x();
162 t[9] = 1.0f / ray.direction.y();
163 t[10] = 1.0f / ray.direction.z();
164
165 t[0] = (box.min.x() - ray.origin.x()) * t[8];
166 t[1] = (box.max.x() - ray.origin.x()) * t[8];
167 t[2] = (box.min.y() - ray.origin.y()) * t[9];
168 t[3] = (box.max.y() - ray.origin.y()) * t[9];
169 t[4] = (box.min.z() - ray.origin.z()) * t[10];
170 t[5] = (box.max.z() - ray.origin.z()) * t[10];
171 t[6] = (float)fmax(fmax(fmin(t[0], t[1]), fmin(t[2], t[3])), fmin(t[4], t[5]));
172 t[7] = (float)fmin(fmin(fmax(t[0], t[1]), fmax(t[2], t[3])), fmax(t[4], t[5]));
173
174 res.hit = !((t[7] < 0) || (t[6] > t[7]));
175 res.distance = t[6];
176 return res;
177}

179inline Vec3 rotateVector(const Vec3& v, Vec3 axis, float angle)
180{
181 // Using Euler-Rodrigues Formula
182 // Ref.: https://en.wikipedia.org/w/index.php?title=Euler%E2%80%93Rodrigues_formula
183
184 Vec3 result = v;
185 axis = axis.normalized();
186
187 angle /= 2;
188 float a = sinf(angle);
189 const float b = axis.x() * a;
190 const float c = axis.y() * a;
191 const float d = axis.z() * a;
192 a = cosf(angle);
193
194 const Vec3 w = {b, c, d};
195 const Vec3 wv = w.cross(v);
196 const Vec3 wwv = w.cross(wv);
197
198 result += wv * 2 * a + wwv * 2;
199
200 return result;
201}
