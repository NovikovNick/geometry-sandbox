78 template <typename Prop, typename SubProp>

80 {
81 if (const reflection::Property* property = reflection::findChildProperty(target.propertyId, next.propertyName); property)
82 {
83 return PropertyPath<Prop, SubProp>{Target{.entityId = target.entityId, .propertyId = property->id}};

85 else

91 template <typename Prop>
92 Key to(PropertyValue<Prop> val) const
93 {
94 Key res;
95 res.target = target;
96 res.rootPropertyId = reflection::getPropertyRoot(target.propertyId).id;
97 res.easing = val.easing == nullptr ? linear::in : val.easing;
98
99 if (val.interpolator)
100 {
101 res.interpolator = [interpolator = val.interpolator](const std::span<float> lhs,
102 const std::span<float> rhs,
103 std::span<float> out,
104 float t)
105 {
106 Prop lhsObj, rhsObj;
107 FloatSpanAdapter<Prop>::toObj(lhs, lhsObj);
108 FloatSpanAdapter<Prop>::toObj(rhs, rhsObj);
109 FloatSpanAdapter<Prop>::toFloatSpan(interpolator(lhsObj, rhsObj, t), out);
110 };
111 }
112
113 res.values.resize(FloatSpanAdapter<Prop>::channelCount);
114 FloatSpanAdapter<Prop>::toFloatSpan(val.value, std::span{res.values});
115
116 return res;
117 }
